# Addition of random real decimals #
for (i = 0; i < 256; i++) {
d = Math.floor(Math.random()*7)+1;
c = Math.floor(Math.random()*7)+1;
p = Math.round(Math.random()*10**d)/10**d;
s = Math.round(Math.random()*10**c)/10**c;
console.log(p + " + " + s + " = " + (p+s).toFixed(13)*1);}

# Subtraction of random real decimals #
for (i = 0; i < 256; i++) {
d = Math.floor(Math.random()*7)+1;
c = Math.floor(Math.random()*7)+1;
p = Math.round(Math.random()*10**d)/10**d;
s = Math.round(Math.random()*10**c)/10**c;
console.log(p + " − " + s + " = " + (p-s).toFixed(13)*1);}

# Product of random real decimals #
for (i = 0; i < 256; i++) {
d = Math.floor(Math.random()*7)+1;
c = Math.floor(Math.random()*7)+1;
p = Math.round(Math.random()*10**d)/10**d;
s = Math.round(Math.random()*10**c)/10**c;
console.log(p + " * " + s + " = " + (p*s).toFixed(14)*1);}

# Square of random real decimal #
for (i = 0; i < 256; i++) {
d = Math.floor(Math.random()*7)+1;
p = Math.round(Math.random()*10**d)/10**d;
console.log("(" + p + ")² = " + (p**2).toFixed(14)*1);}
